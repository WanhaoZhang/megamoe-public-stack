cmake_minimum_required(VERSION 3.18)
project(deepep_clock_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ASCEND_HOME_PATH)
    set(ASCEND_HOME_PATH "$ENV{ASCEND_HOME_PATH}")
endif()
if(NOT ASCEND_HOME_PATH OR NOT EXISTS "${ASCEND_HOME_PATH}")
    message(FATAL_ERROR "Set ASCEND_HOME_PATH to the existing CANN installation")
endif()
execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -v OUTPUT_VARIABLE VER_OUT ERROR_VARIABLE VER_ERR)
string(REGEX MATCH "([0-9][0-9][0-9][0-9])-([0-9][0-9])-([0-9][0-9])" BUILD_DATE "${VER_OUT} ${VER_ERR}")
string(REPLACE "-" "" BUILD_STAMP "${BUILD_DATE}")
set(INCLUDE_ROOTS "${ASCEND_HOME_PATH}" "${ASCEND_HOME_PATH}/${CMAKE_SYSTEM_PROCESSOR}-linux")
get_filename_component(ASCEND_REAL "${ASCEND_HOME_PATH}" REALPATH)
list(APPEND INCLUDE_ROOTS "${ASCEND_REAL}")
set(PROBE_INCLUDES)
set(PROBE_LINKS)
foreach(ROOT IN LISTS INCLUDE_ROOTS)
    list(APPEND PROBE_INCLUDES "${ROOT}/include" "${ROOT}/include/experiment/runtime"
        "${ROOT}/pkg_inc/runtime" "${ROOT}/pkg_inc/runtime/runtime"
        "${ROOT}/compiler/tikcpp" "${ROOT}/compiler/tikcpp/tikcfw"
        "${ROOT}/compiler/tikcpp/tikcfw/impl" "${ROOT}/compiler/tikcpp/tikcfw/interface"
        "${ROOT}/asc/include")
    list(APPEND PROBE_LINKS "${ROOT}/lib64")
endforeach()
if(ASCEND_DEEPEP_RUNTIME_INCLUDE_DIR)
    list(APPEND PROBE_INCLUDES "${ASCEND_DEEPEP_RUNTIME_INCLUDE_DIR}")
endif()
add_library(deepep_clock_probe SHARED clock_probe.cpp)
target_include_directories(deepep_clock_probe PRIVATE ${PROBE_INCLUDES})
target_link_directories(deepep_clock_probe PRIVATE ${PROBE_LINKS})
if(BUILD_STAMP AND BUILD_STAMP GREATER_EQUAL 20260327)
    target_compile_definitions(deepep_clock_probe PRIVATE ASCEND_DEEPEP_USE_NEW_ASCENDC_FRONTEND=1)
    target_compile_options(deepep_clock_probe PRIVATE -O2 -xasc --npu-arch=dav-3510
        -Xhost-start -ftrapv -D__ASC_NPU_HOST__ -Xhost-end --cce-auto-infer-kernel-type=false)
else()
    target_compile_options(deepep_clock_probe PRIVATE -O2 -xcce --cce-aicore-arch=dav-c310
        -Xhost-start -ftrapv -Xhost-end)
    target_link_options(deepep_clock_probe PRIVATE --cce-aicore-arch=dav-c310)
endif()
target_link_options(deepep_clock_probe PRIVATE --cce-fatobj-link)
target_link_libraries(deepep_clock_probe PRIVATE runtime ascendcl stdc++ m)
